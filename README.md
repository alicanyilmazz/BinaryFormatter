SELECT
    DATALENGTH(RESPONSE_DATA) AS SizeBytes,
    DATALENGTH(RESPONSE_DATA) / 1048576.0 AS SizeMB,
    LEFT(RESPONSE_DATA, 200) AS First200,
    RIGHT(RESPONSE_DATA, 500) AS Last500,
    ISJSON(RESPONSE_DATA) AS IsValidJson
FROM ATM_RESPONSE_LOG
WHERE ID = 123;


SELECT
    DATALENGTH(RESPONSE_DATA) AS TotalBytes,
    RIGHT(RESPONSE_DATA, 300) AS LastCharacters,
    ISJSON(RESPONSE_DATA) AS IsValidJson
FROM ATM_RESPONSE_LOG
WHERE ID = 123;
